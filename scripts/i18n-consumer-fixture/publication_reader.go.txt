package main

import (
	"bytes"
	"crypto/sha256"
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"os"
	"path/filepath"
	"slices"
	"strings"

	"github.com/frostgrove/vv/i18n"
)

const (
	pointerLimit = 64 << 10
	fileLimit    = 128 << 20
)

type publicationFile struct {
	Role   string `json:"role"`
	Name   string `json:"name"`
	Bytes  int64  `json:"bytes"`
	SHA256 string `json:"sha256"`
}

type publicationPointer struct {
	Schema     string            `json:"schema"`
	Generation string            `json:"generation"`
	Address    string            `json:"address"`
	Files      []publicationFile `json:"files"`
}

func main() {
	if len(os.Args) != 2 {
		fmt.Fprintln(os.Stderr, "usage: publication-reader ROOT")
		os.Exit(2)
	}
	if err := verifyPublication(os.Args[1]); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

func verifyPublication(rootPath string) error {
	root, err := openPublicationRoot(rootPath)
	if err != nil {
		return err
	}
	defer root.Close()
	raw, err := readRegular(root, "current.json", pointerLimit)
	if err != nil {
		return err
	}
	var pointer publicationPointer
	decoder := json.NewDecoder(bytes.NewReader(raw))
	decoder.DisallowUnknownFields()
	if err := decoder.Decode(&pointer); err != nil {
		return err
	}
	if err := ensureJSONEnd(decoder); err != nil {
		return err
	}
	canonical, err := json.MarshalIndent(pointer, "", "  ")
	if err != nil {
		return err
	}
	canonical = append(canonical, '\n')
	if !bytes.Equal(raw, canonical) {
		return errors.New("publication pointer is not canonical")
	}
	if pointer.Schema != "frostgrove.i18n.publication/v1" || !validAddress(pointer.Generation) || !validAddress(pointer.Address) {
		return errors.New("publication pointer identity is invalid")
	}
	want := []publicationFile{
		{Role: "manifest", Name: "messages.public.json"},
		{Role: "typescript", Name: "messages.d.ts"},
	}
	if len(pointer.Files) != len(want) {
		return errors.New("publication file table is invalid")
	}
	generations, err := openDirectory(root, "generations")
	if err != nil {
		return err
	}
	defer generations.Close()
	generation, err := openDirectory(generations, "sha256-"+strings.TrimPrefix(pointer.Generation, "sha256:"))
	if err != nil {
		return err
	}
	defer generation.Close()
	names, err := readDirectoryNames(generation, len(want))
	if err != nil {
		return err
	}
	if !slices.Equal(names, []string{"messages.d.ts", "messages.public.json"}) {
		return errors.New("publication generation file set is invalid")
	}
	contents := make(map[string][]byte, len(pointer.Files))
	for index, file := range pointer.Files {
		if file.Role != want[index].Role || file.Name != want[index].Name || file.Bytes < 0 || file.Bytes > fileLimit || !validAddress(file.SHA256) {
			return errors.New("publication file descriptor is invalid")
		}
		content, err := readRegular(generation, file.Name, fileLimit)
		if err != nil {
			return err
		}
		if int64(len(content)) != file.Bytes || address(content) != file.SHA256 {
			return fmt.Errorf("publication file %q is invalid", file.Name)
		}
		contents[file.Role] = content
	}
	if i18n.ExpectedPublicExportAddress(contents["manifest"], contents["typescript"]) != pointer.Address {
		return errors.New("publication public export address is invalid")
	}
	if generationAddress(pointer.Address, pointer.Files, contents) != pointer.Generation {
		return errors.New("publication generation address is invalid")
	}
	return nil
}

func openPublicationRoot(path string) (*os.Root, error) {
	if path == "" {
		return nil, errors.New("publication root is empty")
	}
	absolute, err := filepath.Abs(path)
	if err != nil {
		return nil, err
	}
	absolute = filepath.Clean(absolute)
	volumeRoot := filepath.VolumeName(absolute) + string(filepath.Separator)
	relative, err := filepath.Rel(volumeRoot, absolute)
	if err != nil {
		return nil, err
	}
	root, err := os.OpenRoot(volumeRoot)
	if err != nil {
		return nil, err
	}
	if relative == "." {
		return root, nil
	}
	for _, component := range strings.Split(relative, string(filepath.Separator)) {
		info, err := root.Lstat(component)
		if err != nil {
			_ = root.Close()
			return nil, err
		}
		if info.Mode()&os.ModeSymlink != 0 || !info.Mode().IsDir() {
			_ = root.Close()
			return nil, errors.New("publication root path contains a non-directory or symbolic link")
		}
		next, err := root.OpenRoot(component)
		if err != nil {
			_ = root.Close()
			return nil, err
		}
		opened, err := next.Stat(".")
		if err != nil || !os.SameFile(info, opened) {
			_ = next.Close()
			_ = root.Close()
			return nil, errors.New("publication root path changed while opening")
		}
		if err := root.Close(); err != nil {
			_ = next.Close()
			return nil, err
		}
		root = next
	}
	return root, nil
}

func openDirectory(root *os.Root, name string) (*os.Root, error) {
	info, err := root.Lstat(name)
	if err != nil {
		return nil, err
	}
	if info.Mode()&os.ModeSymlink != 0 || !info.Mode().IsDir() {
		return nil, fmt.Errorf("publication path %q is not a regular directory", name)
	}
	directory, err := root.OpenRoot(name)
	if err != nil {
		return nil, err
	}
	opened, err := directory.Stat(".")
	if err != nil || !os.SameFile(info, opened) {
		_ = directory.Close()
		return nil, fmt.Errorf("publication directory %q changed while opening", name)
	}
	return directory, nil
}

func readDirectoryNames(root *os.Root, maximum int) ([]string, error) {
	directory, err := root.Open(".")
	if err != nil {
		return nil, err
	}
	names, readErr := directory.Readdirnames(maximum + 1)
	closeErr := directory.Close()
	if readErr != nil && !errors.Is(readErr, io.EOF) || closeErr != nil {
		return nil, errors.Join(readErr, closeErr)
	}
	if len(names) > maximum {
		return nil, errors.New("publication generation exceeds the file limit")
	}
	slices.Sort(names)
	return names, nil
}

func readRegular(root *os.Root, name string, maximum int64) ([]byte, error) {
	info, err := root.Lstat(name)
	if err != nil {
		return nil, err
	}
	if info.Mode()&os.ModeSymlink != 0 || !info.Mode().IsRegular() || info.Size() > maximum {
		return nil, fmt.Errorf("%q is not a bounded regular file", name)
	}
	file, err := root.Open(name)
	if err != nil {
		return nil, err
	}
	opened, err := file.Stat()
	if err != nil || !opened.Mode().IsRegular() || !os.SameFile(info, opened) {
		_ = file.Close()
		return nil, fmt.Errorf("publication file %q changed while opening", name)
	}
	content, readErr := io.ReadAll(io.LimitReader(file, maximum+1))
	closeErr := file.Close()
	if readErr != nil || closeErr != nil {
		return nil, errors.Join(readErr, closeErr)
	}
	if int64(len(content)) > maximum {
		return nil, fmt.Errorf("%q exceeds its byte limit", name)
	}
	return content, nil
}

func ensureJSONEnd(decoder *json.Decoder) error {
	var extra any
	if err := decoder.Decode(&extra); !errors.Is(err, io.EOF) {
		if err == nil {
			return errors.New("publication pointer has trailing data")
		}
		return err
	}
	return nil
}

func validAddress(value string) bool {
	if len(value) != len("sha256:")+sha256.Size*2 || !strings.HasPrefix(value, "sha256:") {
		return false
	}
	digest := strings.TrimPrefix(value, "sha256:")
	decoded, err := hex.DecodeString(digest)
	return err == nil && len(decoded) == sha256.Size && strings.ToLower(digest) == digest
}

func address(content []byte) string {
	digest := sha256.Sum256(content)
	return "sha256:" + hex.EncodeToString(digest[:])
}

func generationAddress(publicAddress string, files []publicationFile, contents map[string][]byte) string {
	hash := sha256.New()
	writeField(hash, "domain", []byte("frostgrove.i18n.publication-generation/v1"))
	writeField(hash, "address", []byte(publicAddress))
	for _, file := range files {
		writeField(hash, "role", []byte(file.Role))
		writeField(hash, "name", []byte(file.Name))
		writeField(hash, "content", contents[file.Role])
	}
	return "sha256:" + hex.EncodeToString(hash.Sum(nil))
}

func writeField(destination io.Writer, name string, value []byte) {
	var size [8]byte
	binary.BigEndian.PutUint64(size[:], uint64(len(name)))
	_, _ = destination.Write(size[:])
	_, _ = io.WriteString(destination, name)
	binary.BigEndian.PutUint64(size[:], uint64(len(value)))
	_, _ = destination.Write(size[:])
	_, _ = destination.Write(value)
}
